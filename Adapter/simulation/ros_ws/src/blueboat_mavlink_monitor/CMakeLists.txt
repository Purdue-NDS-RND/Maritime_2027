cmake_minimum_required(VERSION 3.8)
project(blueboat_mavlink_monitor)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(mavlink_v2_monitor src/mavlink_v2_monitor.cpp)
target_include_directories(mavlink_v2_monitor PRIVATE include)
ament_target_dependencies(mavlink_v2_monitor rclcpp std_msgs)
install(TARGETS mavlink_v2_monitor DESTINATION lib/${PROJECT_NAME})
ament_package()
