cmake_minimum_required(VERSION 3.16)
project(maritime_buoy_systems LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(BuoyFaceBoat SHARED BuoyFaceBoat.cc)
target_compile_features(BuoyFaceBoat PRIVATE cxx_std_17)
target_compile_options(BuoyFaceBoat PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(BuoyFaceBoat PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS BuoyFaceBoat LIBRARY DESTINATION lib)
add_library(RunEvidence SHARED RunEvidence.cc)
target_compile_features(RunEvidence PRIVATE cxx_std_17)
target_compile_options(RunEvidence PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(RunEvidence PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS RunEvidence LIBRARY DESTINATION lib)
